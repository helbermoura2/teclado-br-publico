cmake_minimum_required(VERSION 3.22)
project(tecladoai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
foreach(feature LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_OPENSSL LLAMA_SUBPROCESS GGML_OPENMP GGML_NATIVE GGML_BACKEND_DL GGML_LLAMAFILE)
  set(${feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(llama.cpp)
add_library(tecladoai SHARED engine.cpp)
target_link_libraries(tecladoai PRIVATE llama-common llama log)
target_link_options(tecladoai PRIVATE "-Wl,-z,max-page-size=16384")
